def upload_to_pastefy(content: str) -> str:
    """رفع المحتوى إلى Pastefy"""
    try:
        header = "DUMPER is bot HROXEN\nTELEGRAM @HROXEN_FOR_DUMPER_BOT\n\n"
        full_content = header + content
        
        files = {'file': ('file.txt', full_content)}
        response = requests.post('https://pastefy.app', files=files, timeout=30)
        if response.status_code == 200:
            paste_id = response.text.strip()
            match = re.search(r'pastefy\.app/([a-zA-Z0-9]+)', paste_id)
            if match:
                paste_id = match.group(1)
            return f"https://pastefy.app/{paste_id}/raw"
        return None
    except Exception as e:
        logger.error(f"خطأ في رفع Pastefy: {e}")
        return None


def decrypt_encryptx_script(script: str, key: str) -> str:
    """فك تشفير سكربت Encrypt X"""
    try:
        padding = 8 - (len(script) % 8)
        if padding != 8:
            script += "=" * padding
        
        data = base64.b32decode(script)
        key_bytes = key.encode('utf-8')
        result = bytes([data[i] ^ key_bytes[i % len(key_bytes)] for i in range(len(data))])
        
        return result.decode('utf-8', errors='ignore')
    except Exception as e:
        logger.error(f"خطأ في فك التشفير: {e}")
        return f"[Error: {e}]"


def extract_id_from_url(url: str) -> str:
    """استخراج ID من الرابط"""
    match = re.search(r'[?&]Id=([^&]+)', url)
    return match.group(1) if match else None


def get_encryptx_data(script_id: str) -> dict:
    """جلب بيانات Encrypt X"""
    urls = [
        f"https://raw.githubusercontent.com/ScriptObfuscator2/Scripts/main/{script_id}",
        f"https://proxy.2677929.xyz/https://raw.githubusercontent.com/ScriptObfuscator2/Scripts/main/{script_id}"
    ]
    
    for url in urls:
        try:
            response = requests.get(url, timeout=15)
            if response.status_code == 200:
                try:
                    data = json.loads(response.text)
                    if 'Key' in data and 'Script' in data:
                        return data
                except:
                    return {'Script': response.text.strip()}
        except Exception as e:
            logger.error(f"خطأ في جلب البيانات: {e}")
            continue
    
    return None


def generate_random_filename() -> str:
    """توليد اسم ملف عشوائي"""
    letters = string.ascii_lowercase + string.digits
    return ''.join(random.choice(letters) for _ in range(12))


def save_script_to_file(script: str, script_id: str) -> str:
    """حفظ السكربت في ملف"""

    filepath = OUTPUT_DIR / filename

        f.write("=" * 60 + "\n\n")
        f.write(script + "\n\n")
        f.write("=" * 60 + "\n")
        f.write("تم تفكيك تشفير\n")
        f.write(f"ID: {script_id}\n")
        f.write(f"التاريخ: {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}\n")
        f.write("=" * 60)
    
    return str(filepath)

class ResultView(discord.ui.View):
    def __init__(self, pastefy_url, filepath, filename):
        super().__init__(timeout=300)
        self.pastefy_url = pastefy_url
        self.filepath = filepath
        self.filename = filename
        
        # إضافة زر Pastefy (مع url)
        if pastefy_url:
            self.add_item(
                discord.ui.Button(
                    label="محرر",
                    style=discord.ButtonStyle.link,
                    url=pastefy_url
                )
            )
    
    @discord.ui.button(label=" تحميل الملف", style=discord.ButtonStyle.green)
    async def download_button(self, interaction: discord.Interaction, button: discord.ui.Button):
        try:
            with open(self.filepath, 'rb') as f:
                await interaction.response.send_message(
                    file=discord.File(f, self.filename),
                    ephemeral=True
                )
        except Exception as e:
            await interaction.response.send_message(f" خطأ: {str(e)}", ephemeral=True)
    
    @discord.ui.button(label=" نسخ النص", style=discord.ButtonStyle.blurple)
    async def copy_button(self, interaction: discord.Interaction, button: discord.ui.Button):
        try:
            with open(self.filepath, 'r', encoding='utf-8') as f:
                content = f.read()
                if len(content) > 1900:
                    await interaction.response.send_message(
                        "❌ النص طويل جداً! استخدم زر التحميل",
                        ephemeral=True
                    )
                else:
                    await interaction.response.send_message(
                        f"```\n{content[:1900]}\n```",
                        ephemeral=True
                    )
        except Exception as e:
            await interaction.response.send_message(f" خطأ: {str(e)}", ephemeral=True)
